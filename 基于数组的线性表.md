template <typename E> //List ADT 
class List{
private:
    void operator=(const List&){} 	//禁止拷贝构造
    List(const List&){}  	//禁止拷贝赋值
public:
    List(){}    //构造函数
    virtual ~List(){}	//析构函数
    
    //清空列表元素
    virtual void clear()=0;
    
    //插入一个元素到指定位置
    //item是待插入的元素
    virtual void insert(const E& item) = 0;
    
    //将元素添加到列表末端
    //item是待添加的元素
    virtual void append(const E& item) = 0;
    
    //在列表中删除元素
    //返回被删除的元素
    virtual E remove() = 0;
    
    //将指针移动至开头
    virtual void moveToStart() = 0;
    
    //将指针移动至末尾
    virtual void moveToEnd() = 0;
    
    //将指针左移一格（如果在最左边则不动）
    virtual void prev() = 0;
    
    //将指针右移一格（如果在最右边则不动）
    virtual void next() = 0;
    
    //返回列表中元素的个数
    virtual int length() const = 0;
    
    //返回指针所在位置
    virtual int currPos() const = 0;
    
    //将指针移动到pos位置
    virtual void moveToPos(int pos) = 0;
    
    //返回指针指向元素的值
    virtual const E& getValue() const = 0;
};

template <typename E> 
class Alist : public List<E> {  //基于数组的线性表实现
private :
    int maxSize;		//列表可容纳最多元素
    int listSize;		//当前列表中元素个数
    int curr;			//指针
    E* listArray;		//存储线性表实际元素的数组
public:
    Alist(int size=defaultSize){ //构造函数
        maxSize = size;
        listSize = curr = 0;
        listArray = new E[maxSize];
    }
    
    ~Alist(){delete [] listArray;} //析构函数
    
    void clear(){
        delete []listArray; //释放原有数组的内存
        listSize = curr = 0; //重置数组
        listArray = new E[maxSize]; //重新申请数组内存
    }
    
    void insert(const E& it){ //将E类型的it插入到数组中
        assert(listSize<maxSize&&"List capacity exceeded"); //检查前面的情况成立，															  //否则报错文字
        for(int i=listSize;i>curr;i--)
            listArray[i] = listArray[i-1];
        listArray[curr] = it;
        listSize++;
    }
    
    void append(const E& it){ //将E类型的it放在末尾
        assert(listSize < maxSize&&"List capacity exceeded");
        listArray[listSize++] = it;
    }
    
    E remove(){ //将指针所指的数删除并返回删除的数
        assert((curr>=0) && (curr < listSize)&&"No element");
        E it =listArray[curr];
        for(int i = curr;i<listSize-1;i++)
            listArray[i]=listArray[i+1];
        listSize--;
        return it;
    }
    
    void moveToStart() { curr = 0; } //将指针指向开头
    void moveToEnd() { if(listSize) curr = listSize-1; } //将指针指向末尾
    void prev() { if (curr != 0) curr--; } //指针左移一位
    void next() { if (curr < listSize) curr++; } //指针右移一位
    
    int length() const {return listSize; } //获取数组长度
    
    int currPos() const {return curr; } //获取指针位置
    
    void moveToPos(int pos){ //将指针移动到指定位置
        assert ((pos>=0)&&(pos<=listSize)&&"Pos out of range");
        curr = pos;
    }
    
    const E& getValue() const { //获取指针指向的值
        assert((curr>=0)&&(curr<listSize)&&"No current element");
        return listArray[curr];
    }
};
    
//这样我们就有了一个可自己定义并操作的数组了
